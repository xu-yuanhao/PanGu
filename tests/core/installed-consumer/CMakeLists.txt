cmake_minimum_required(VERSION 3.20)
project(PanGuInstalledConsumer LANGUAGES C)
find_package(PanGuCore CONFIG REQUIRED)
get_target_property(core_links PanGu::core INTERFACE_LINK_LIBRARIES)
if(core_links MATCHES "CURL|tui|services|agent")
  message(FATAL_ERROR "Core package unexpectedly imports product dependencies: ${core_links}")
endif()
add_executable(consumer main.c)
set_target_properties(consumer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_link_libraries(consumer PRIVATE PanGu::core)
enable_testing()
add_test(NAME installed_embed COMMAND consumer)
