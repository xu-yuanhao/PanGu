cmake_minimum_required(VERSION 3.20)
project(PanGuInstalledTUI LANGUAGES C)
find_package(PanGuTUI CONFIG REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(tui_consumer main.c)
target_link_libraries(tui_consumer PRIVATE PanGu::tui)
set_target_properties(tui_consumer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
enable_testing()
add_test(NAME installed_tui_terminal COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/terminal.py" $<TARGET_FILE:tui_consumer>)
if(PANGU_CONSUMER_PRESENTATION)
  find_package(PanGuTUIPresentation CONFIG REQUIRED)
  add_executable(presentation_consumer presentation.c)
  target_link_libraries(presentation_consumer PRIVATE PanGu::tui_plugin PanGu::agent_app)
  set_target_properties(presentation_consumer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  add_test(NAME installed_tui_presentation COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/terminal.py" $<TARGET_FILE:presentation_consumer> "${CMAKE_CURRENT_BINARY_DIR}/session")
  add_test(NAME installed_tui_legacy_config COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/terminal.py" $<TARGET_FILE:presentation_consumer> "${CMAKE_CURRENT_BINARY_DIR}/legacy-session" legacy)
  add_test(NAME installed_tui_direct_legacy COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/terminal.py" $<TARGET_FILE:presentation_consumer> "${CMAKE_CURRENT_BINARY_DIR}/direct-session" direct)
endif()
