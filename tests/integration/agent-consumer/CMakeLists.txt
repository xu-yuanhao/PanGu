cmake_minimum_required(VERSION 3.20)
project(PanGuInstalledAgent LANGUAGES C)
find_package(PanGuAgent CONFIG REQUIRED)
add_executable(agent_consumer main.c)
target_link_libraries(agent_consumer PRIVATE PanGu::agent_app)
set_target_properties(agent_consumer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
enable_testing()
if(TARGET PanGu::services_mcp_http)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(app_http_consumer mcp-http.c)
  target_link_libraries(app_http_consumer PRIVATE PanGu::agent_app PanGu::services_mcp_http PanGu::services_schema)
  set_target_properties(app_http_consumer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  foreach(backend jsonl sqlite)
    add_test(NAME installed_app_http_${backend} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../services/mcp-http.py" $<TARGET_FILE:app_http_consumer> --installed)
    set_tests_properties(installed_app_http_${backend} PROPERTIES ENVIRONMENT "PANGU_HTTP_STORE=${backend}")
  endforeach()
endif()
add_executable(approval_consumer approval.c)
target_link_libraries(approval_consumer PRIVATE PanGu::agent_app)
set_target_properties(approval_consumer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
add_library(consumer_model MODULE plugin.c)
target_include_directories(consumer_model PRIVATE "$<TARGET_PROPERTY:PanGu::agent_app,INTERFACE_INCLUDE_DIRECTORIES>")
set_target_properties(consumer_model PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO C_VISIBILITY_PRESET hidden)
foreach(backend jsonl sqlite)
  add_test(NAME approval_consumer_${backend} COMMAND approval_consumer "${CMAKE_CURRENT_BINARY_DIR}/approval-${backend}" ${backend})
  add_test(NAME agent_consumer_${backend} COMMAND agent_consumer "${CMAKE_CURRENT_BINARY_DIR}/session-${backend}" ${backend} $<TARGET_FILE:consumer_model>)
endforeach()
find_program(installed_cli NAMES pangu_agent_cli PATHS ${CMAKE_PREFIX_PATH} PATH_SUFFIXES bin NO_DEFAULT_PATH REQUIRED)
set(mcp_arguments)
if(TARGET PanGu::services_mcp AND NOT WIN32)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  list(APPEND mcp_arguments "-Dpython=${Python3_EXECUTABLE}" "-Dpeer=${CMAKE_CURRENT_SOURCE_DIR}/../services-consumer/mcp-server.py")
endif()
add_test(NAME installed_cli_reopen COMMAND ${CMAKE_COMMAND} "-Dcli=${installed_cli}" "-Dsession=${CMAKE_CURRENT_BINARY_DIR}/cli-session" "-Dplugin=$<TARGET_FILE:consumer_model>" ${mcp_arguments} -P "${CMAKE_CURRENT_SOURCE_DIR}/cli.cmake")
