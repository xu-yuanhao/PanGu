cmake_minimum_required(VERSION 3.20)
project(PanGuInstalledServices LANGUAGES C)
find_package(PanGuServices CONFIG REQUIRED)
add_executable(services_consumer main.c)
target_link_libraries(services_consumer PRIVATE PanGu::services PanGu::services_schema PanGu::services_deepseek)
set_target_properties(services_consumer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
enable_testing()
if(TARGET PanGu::services_mcp_http)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(mcp_http_consumer mcp-http.c)
  target_link_libraries(mcp_http_consumer PRIVATE PanGu::services_mcp_http PanGu::services_schema)
  set_target_properties(mcp_http_consumer PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  add_test(NAME installed_mcp_http COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../services/mcp-http.py" $<TARGET_FILE:mcp_http_consumer> --installed)
endif()
add_library(consumer_plugin MODULE plugin.c)
target_include_directories(consumer_plugin PRIVATE "$<TARGET_PROPERTY:PanGu::services,INTERFACE_INCLUDE_DIRECTORIES>")
set_target_properties(consumer_plugin PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO C_VISIBILITY_PRESET hidden)
add_library(consumer_legacy_plugin MODULE plugin.c)
target_include_directories(consumer_legacy_plugin PRIVATE "$<TARGET_PROPERTY:PanGu::services,INTERFACE_INCLUDE_DIRECTORIES>")
target_compile_definitions(consumer_legacy_plugin PRIVATE PANGU_TEST_LEGACY_ABI=1)
set_target_properties(consumer_legacy_plugin PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO C_VISIBILITY_PRESET hidden)
set(consumer_arguments $<TARGET_FILE:consumer_plugin> $<TARGET_FILE:consumer_legacy_plugin>)
if(TARGET PanGu::services_mcp AND NOT WIN32)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  target_link_libraries(services_consumer PRIVATE PanGu::services_mcp)
  target_compile_definitions(services_consumer PRIVATE PANGU_CONSUMER_MCP=1)
  list(APPEND consumer_arguments "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/mcp-server.py")
endif()
add_test(NAME services_consumer COMMAND services_consumer ${consumer_arguments})

if(TARGET PanGu::services_resources)
  target_link_libraries(services_consumer PRIVATE PanGu::services_resources)
  target_compile_definitions(services_consumer PRIVATE PANGU_CONSUMER_RESOURCES=1)
  set(resource_fixture "${CMAKE_CURRENT_BINARY_DIR}/resources")
  file(MAKE_DIRECTORY "${resource_fixture}/user/prompts" "${resource_fixture}/user/skills/kept" "${resource_fixture}/user/skills/ignored" "${resource_fixture}/project")
  file(WRITE "${resource_fixture}/user/settings.json" "{\"defaultModel\":\"consumer-model\"}")
  file(WRITE "${resource_fixture}/user/prompts/hello.md" "---\ndescription: Installed template\n---\nHello $1")
  file(WRITE "${resource_fixture}/user/skills/.gitignore" "ignored/\n")
  file(WRITE "${resource_fixture}/user/skills/kept/SKILL.md" "---\nname: kept\ndescription: Installed skill\n---\nRetained skill")
  file(WRITE "${resource_fixture}/user/skills/ignored/SKILL.md" "---\nname: ignored\ndescription: Hidden skill\n---\nIgnored skill")
  set_tests_properties(services_consumer PROPERTIES ENVIRONMENT "PANGU_CONSUMER_RESOURCES=${resource_fixture}")
endif()
