# Source overflow classification is optional and keeps PCRE2 out of base Core.
option(PANGU_BUILD_OVERFLOW "Build optional source overflow classifier" OFF)
if(PANGU_BUILD_OVERFLOW)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PANGU_PCRE2 REQUIRED IMPORTED_TARGET libpcre2-8>=10.43)
  add_library(pangu_overflow src/core/overflow.c)
  add_library(PanGu::overflow ALIAS pangu_overflow)
  target_link_libraries(pangu_overflow PUBLIC PanGu::core PRIVATE PkgConfig::PANGU_PCRE2)
  set_target_properties(pangu_overflow PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO EXPORT_NAME overflow)
  if(NOT MSVC)
    target_compile_options(pangu_overflow PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endif()
