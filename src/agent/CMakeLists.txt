# Session and app share one SDK export. The app requires the services host;
# presentation and production providers belong to the CLI composition only.
# Static PRIVATE libraries remain LINK_ONLY dependencies in the export. The
# parent-owned PanGuAgentConfig must create the same imported dependency names.
install(TARGETS pangu_agent pangu_agent_app EXPORT PanGuAgentTargets
 ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
 LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
 RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS pangu_agent_cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY "${PANGU_AGENT_ROOT}/include/pangu/agent"
 DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/pangu FILES_MATCHING PATTERN "*.h")
install(EXPORT PanGuAgentTargets FILE PanGuAgentTargets.cmake
 NAMESPACE PanGu:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PanGuAgent)
